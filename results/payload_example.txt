%%bash
cat << 'EOF' > deseq2_analysis.R

# Load required packages
suppressPackageStartupMessages({
  library(DESeq2)
  library(dplyr)
  library(tibble)
})

# Load data
cat("Loading count data and metadata...\n")
count_data <- read.csv("count_data_filtered.csv", row.names=1, check.names=FALSE)
coldata <- read.csv("coldata.csv")

cat("Count data dimensions:", dim(count_data), "\n")
cat("Metadata dimensions:", dim(coldata), "\n")

# Ensure sample order matches
coldata <- coldata[match(colnames(count_data), coldata$sample), ]
cat("Sample order check - all match:", all(coldata$sample == colnames(count_data)), "\n")

# Set condition as factor with Control as reference
coldata$condition <- factor(coldata$condition, levels = c("Control", "ASXL1"))

cat("Condition table:\n")
print(table(coldata$condition))

# Create DESeq2 dataset
cat("\nCreating DESeq2 dataset...\n")
dds <- DESeqDataSetFromMatrix(
  countData = count_data,
  colData = coldata,
  design = ~ condition
)

# Pre-filter: remove rows with low counts
keep <- rowSums(counts(dds)) >= 10
dds <- dds[keep, ]
cat("Genes after filtering (rowSums >= 10):", nrow(dds), "\n")

# Run DESeq2
cat("\nRunning DESeq2...\n")
dds <- DESeq(dds)

# Get results
res <- results(dds, contrast = c("condition", "ASXL1", "Control"))
res_df <- as.data.frame(res) %>%
  tibble::rownames_to_column("gene_id")

cat("\nDESeq2 results summary:\n")
summary(res)

# Filter significant DEGs
sig_degs <- res_df %>%
  filter(!is.na(padj) & padj < 0.05)

cat("\nNumber of significant DEGs (padj < 0.05):", nrow(sig_degs), "\n")
cat("Upregulated (log2FC > 0):", sum(sig_degs$log2FoldChange > 0), "\n")
cat("Downregulated (log2FC < 0):", sum(sig_degs$log2FoldChange < 0), "\n")

# Save results
write.csv(res_df, "deseq2_results.csv", row.names=FALSE)
write.csv(sig_degs, "deseq2_sig_degs.csv", row.names=FALSE)
cat("\nResults saved!\n")

EOF

# Load required packages
suppressPackageStartupMessages({
  library(clusterProfiler)
  library(org.Hs.eg.db)
  library(dplyr)
  library(tibble)
})

# Load significant DEGs
cat("Loading significant DEGs...\n")
sig_degs <- read.csv("deseq2_sig_degs.csv")
cat("Number of significant DEGs:", nrow(sig_degs), "\n")
cat("Head of gene IDs:\n")
print(head(sig_degs$gene_id))

# The gene IDs appear to be ENSEMBL IDs with version numbers (e.g., ENSG00000223972.5)
# Remove version numbers for ID mapping
sig_degs$gene_id_clean <- sub("\\..*", "", sig_degs$gene_id)
cat("\nCleaned gene IDs (first 5):\n")
print(head(sig_degs$gene_id_clean))

# Convert ENSEMBL IDs to ENTREZ IDs for enrichGO
cat("\nConverting ENSEMBL IDs to ENTREZ IDs...\n")
entrez_ids <- bitr(sig_degs$gene_id_clean, 
                   fromType = "ENSEMBL", 
                   toType = "ENTREZID", 
                   OrgDb = org.Hs.eg.db)

cat("Number of genes with ENTREZ IDs:", nrow(entrez_ids), "\n")
cat("Conversion rate:", round(nrow(entrez_ids)/length(unique(sig_degs$gene_id_clean))*100, 1), "%\n")

# Run enrichGO analysis - Biological Process
cat("\nRunning enrichGO analysis (BP)...\n")
ego <- enrichGO(gene          = entrez_ids$ENTREZID,
                OrgDb         = org.Hs.eg.db,
                ont           = "BP",
                pAdjustMethod = "BH",
                pvalueCutoff  = 0.05,
                qvalueCutoff  = 0.2,
                readable      = TRUE)

cat("Number of enriched GO terms before simplify:", nrow(ego@result), "\n")

# Apply simplify with similarity > 0.7
cat("\nApplying simplify() with cutoff = 0.7...\n")
ego_simplified <- simplify(ego, cutoff = 0.7, by = "p.adjust", select_fun = min)

cat("Number of enriched GO terms after simplify:", nrow(ego_simplified@result), "\n")

# Save the full simplified results
results_df <- as.data.frame(ego_simplified)
write.csv(results_df, "enrichgo_simplified_results.csv", row.names=FALSE)
cat("\nSimplified results saved!\n")

# Show results related to T cell
cat("\nGO terms related to 'T cell':\n")
tcell_results <- results_df %>%
  filter(grepl("T cell", Description, ignore.case = TRUE)) %>%
  dplyr::select(Description, pvalue, p.adjust, qvalue, Count)
print(tcell_results)

EOF

cat("Metadata dimensions:", dim(coldata), "\n")

# Ensure sample order matches

cat("Sample order check - all match:", all(coldata$sample == colnames(count_data)), "\n")

cat("Condition table:\n")
print(table(coldata$condition))

# Create DESeq2 dataset
cat("\nCreating DESeq2 dataset...\n")
dds <- DESeqDataSetFromMatrix(
  countData = count_data,
  colData = coldata,
  design = ~ condition
)

# Pre-filter: remove rows with low counts
keep <- rowSums(counts(dds)) >= 10
dds <- dds[keep, ]
cat("Genes after filtering (rowSums >= 10):", nrow(dds), "\n")

# Run DESeq2
cat("\nRunning DESeq2...\n")
dds <- DESeq(dds)

# Get results
res <- results(dds, contrast = c("condition", "ASXL1", "Control"))
res_df <- as.data.frame(res) %>%
  tibble::rownames_to_column("gene_id")

summary(res)

%%bash
cat << 'EOF' > deseq2_analysis.R

# Load required packages
suppressPackageStartupMessages({
  library(DESeq2)
  library(dplyr)
  library(tibble)
})

# Load data
cat("Loading count data and metadata...\n")
count_data <- read.csv("count_data_filtered.csv", row.names=1, check.names=FALSE)
coldata <- read.csv("coldata.csv")

cat("Count data dimensions:", dim(count_data), "\n")
cat("Metadata dimensions:", dim(coldata), "\n")

# Ensure sample order matches
coldata <- coldata[match(colnames(count_data), coldata$sample), ]
cat("Sample order check - all match:", all(coldata$sample == colnames(count_data)), "\n")

# Set condition as factor with Control as reference
coldata$condition <- factor(coldata$condition, levels = c("Control", "ASXL1"))

cat("Condition table:\n")
print(table(coldata$condition))

# Create DESeq2 dataset
cat("\nCreating DESeq2 dataset...\n")
dds <- DESeqDataSetFromMatrix(
  countData = count_data,
  colData = coldata,
  design = ~ condition
)

# Pre-filter: remove rows with low counts
keep <- rowSums(counts(dds)) >= 10
dds <- dds[keep, ]
cat("Genes after filtering (rowSums >= 10):", nrow(dds), "\n")

# Run DESeq2
cat("\nRunning DESeq2...\n")
dds <- DESeq(dds)

# Get results
res <- results(dds, contrast = c("condition", "ASXL1", "Control"))
res_df <- as.data.frame(res) %>%
  tibble::rownames_to_column("gene_id")

cat("\nDESeq2 results summary:\n")
summary(res)

# Filter significant DEGs
sig_degs <- res_df %>%
  filter(!is.na(padj) & padj < 0.05)

cat("\nNumber of significant DEGs (padj < 0.05):", nrow(sig_degs), "\n")
cat("Upregulated (log2FC > 0):", sum(sig_degs$log2FoldChange > 0), "\n")
cat("Downregulated (log2FC < 0):", sum(sig_degs$log2FoldChange < 0), "\n")

# Save results
write.csv(res_df, "deseq2_results.csv", row.names=FALSE)
write.csv(sig_degs, "deseq2_sig_degs.csv", row.names=FALSE)
cat("\nResults saved!\n")

EOF

%%bash
cat << 'EOF' > enrichgo_analysis.R

# Load required packages
suppressPackageStartupMessages({
  library(clusterProfiler)
  library(org.Hs.eg.db)
  library(dplyr)
  library(tibble)
})

# Load significant DEGs
cat("Loading significant DEGs...\n")
sig_degs <- read.csv("deseq2_sig_degs.csv")
cat("Number of significant DEGs:", nrow(sig_degs), "\n")
cat("Head of gene IDs:\n")
print(head(sig_degs$gene_id))

# The gene IDs appear to be ENSEMBL IDs with version numbers (e.g., ENSG00000223972.5)
# Remove version numbers for ID mapping
sig_degs$gene_id_clean <- sub("\\..*", "", sig_degs$gene_id)
cat("\nCleaned gene IDs (first 5):\n")
print(head(sig_degs$gene_id_clean))

# Convert ENSEMBL IDs to ENTREZ IDs for enrichGO
cat("\nConverting ENSEMBL IDs to ENTREZ IDs...\n")
entrez_ids <- bitr(sig_degs$gene_id_clean, 
                   fromType = "ENSEMBL", 
                   toType = "ENTREZID", 
                   OrgDb = org.Hs.eg.db)

cat("Number of genes with ENTREZ IDs:", nrow(entrez_ids), "\n")
cat("Conversion rate:", round(nrow(entrez_ids)/length(unique(sig_degs$gene_id_clean))*100, 1), "%\n")

# Run enrichGO analysis - Biological Process
cat("\nRunning enrichGO analysis (BP)...\n")
ego <- enrichGO(gene          = entrez_ids$ENTREZID,
                OrgDb         = org.Hs.eg.db,
                ont           = "BP",
                pAdjustMethod = "BH",
                pvalueCutoff  = 0.05,
                qvalueCutoff  = 0.2,
                readable      = TRUE)

cat("Number of enriched GO terms before simplify:", nrow(ego@result), "\n")

# Apply simplify with similarity > 0.7
cat("\nApplying simplify() with cutoff = 0.7...\n")
ego_simplified <- simplify(ego, cutoff = 0.7, by = "p.adjust", select_fun = min)

cat("Number of enriched GO terms after simplify:", nrow(ego_simplified@result), "\n")

# Save the full simplified results
results_df <- as.data.frame(ego_simplified)
write.csv(results_df, "enrichgo_simplified_results.csv", row.names=FALSE)
cat("\nSimplified results saved!\n")

# Show results related to T cell
cat("\nGO terms related to 'T cell':\n")
tcell_results <- results_df %>%
  filter(grepl("T cell", Description, ignore.case = TRUE)) %>%
  dplyr::select(Description, pvalue, p.adjust, qvalue, Count)
print(tcell_results)

EOF

%%bash
cat << 'EOF' > deseq2_analysis.R

# Load required packages
suppressPackageStartupMessages({
  library(DESeq2)
  library(dplyr)
  library(tibble)
})

# Load data
cat("Loading count data and metadata...\n")
count_data <- read.csv("count_data_filtered.csv", row.names=1, check.names=FALSE)
coldata <- read.csv("coldata.csv")

cat("Count data dimensions:", dim(count_data), "\n")
cat("Metadata dimensions:", dim(coldata), "\n")

# Ensure sample order matches
coldata <- coldata[match(colnames(count_data), coldata$sample), ]
cat("Sample order check - all match:", all(coldata$sample == colnames(count_data)), "\n")

# Set condition as factor with Control as reference
coldata$condition <- factor(coldata$condition, levels = c("Control", "ASXL1"))

cat("Condition table:\n")
print(table(coldata$condition))

# Create DESeq2 dataset
cat("\nCreating DESeq2 dataset...\n")
dds <- DESeqDataSetFromMatrix(
  countData = count_data,
  colData = coldata,
  design = ~ condition
)

# Pre-filter: remove rows with low counts
keep <- rowSums(counts(dds)) >= 10
dds <- dds[keep, ]
cat("Genes after filtering (rowSums >= 10):", nrow(dds), "\n")

# Run DESeq2
cat("\nRunning DESeq2...\n")
dds <- DESeq(dds)

# Get results
res <- results(dds, contrast = c("condition", "ASXL1", "Control"))
res_df <- as.data.frame(res) %>%
  tibble::rownames_to_column("gene_id")

cat("\nDESeq2 results summary:\n")
summary(res)

# Filter significant DEGs
sig_degs <- res_df %>%
  filter(!is.na(padj) & padj < 0.05)

cat("\nNumber of significant DEGs (padj < 0.05):", nrow(sig_degs), "\n")
cat("Upregulated (log2FC > 0):", sum(sig_degs$log2FoldChange > 0), "\n")
cat("Downregulated (log2FC < 0):", sum(sig_degs$log2FoldChange < 0), "\n")

# Save results
write.csv(res_df, "deseq2_results.csv", row.names=FALSE)
write.csv(sig_degs, "deseq2_sig_degs.csv", row.names=FALSE)
cat("\nResults saved!\n")

EOF

%%bash
cat << 'EOF' > enrichgo_analysis.R

# Load required packages
suppressPackageStartupMessages({
  library(clusterProfiler)
  library(org.Hs.eg.db)
  library(dplyr)
  library(tibble)
})

# Load significant DEGs
cat("Loading significant DEGs...\n")
sig_degs <- read.csv("deseq2_sig_degs.csv")
cat("Number of significant DEGs:", nrow(sig_degs), "\n")
cat("Head of gene IDs:\n")
print(head(sig_degs$gene_id))

# The gene IDs appear to be ENSEMBL IDs with version numbers (e.g., ENSG00000223972.5)
# Remove version numbers for ID mapping
sig_degs$gene_id_clean <- sub("\\..*", "", sig_degs$gene_id)
cat("\nCleaned gene IDs (first 5):\n")
print(head(sig_degs$gene_id_clean))

# Convert ENSEMBL IDs to ENTREZ IDs for enrichGO
cat("\nConverting ENSEMBL IDs to ENTREZ IDs...\n")
entrez_ids <- bitr(sig_degs$gene_id_clean, 
                   fromType = "ENSEMBL", 
                   toType = "ENTREZID", 
                   OrgDb = org.Hs.eg.db)

cat("Number of genes with ENTREZ IDs:", nrow(entrez_ids), "\n")
cat("Conversion rate:", round(nrow(entrez_ids)/length(unique(sig_degs$gene_id_clean))*100, 1), "%\n")

# Run enrichGO analysis - Biological Process
cat("\nRunning enrichGO analysis (BP)...\n")
ego <- enrichGO(gene          = entrez_ids$ENTREZID,
                OrgDb         = org.Hs.eg.db,
                ont           = "BP",
                pAdjustMethod = "BH",
                pvalueCutoff  = 0.05,
                qvalueCutoff  = 0.2,
                readable      = TRUE)

cat("Number of enriched GO terms before simplify:", nrow(ego@result), "\n")

# Apply simplify with similarity > 0.7
cat("\nApplying simplify() with cutoff = 0.7...\n")
ego_simplified <- simplify(ego, cutoff = 0.7, by = "p.adjust", select_fun = min)

cat("Number of enriched GO terms after simplify:", nrow(ego_simplified@result), "\n")

# Save the full simplified results
results_df <- as.data.frame(ego_simplified)
write.csv(results_df, "enrichgo_simplified_results.csv", row.names=FALSE)
cat("\nSimplified results saved!\n")

# Show results related to T cell
cat("\nGO terms related to 'T cell':\n")
tcell_results <- results_df %>%
  filter(grepl("T cell", Description, ignore.case = TRUE)) %>%
  dplyr::select(Description, pvalue, p.adjust, qvalue, Count)
print(tcell_results)

EOF

# Load required packages
suppressPackageStartupMessages({
  library(DESeq2)
  library(dplyr)
  library(tibble)
})

# Load data
cat("Loading count data and metadata...\n")
count_data <- read.csv("count_data_filtered.csv", row.names=1, check.names=FALSE)
coldata <- read.csv("coldata.csv")

cat("Count data dimensions:", dim(count_data), "\n")
cat("Metadata dimensions:", dim(coldata), "\n")

# Ensure sample order matches
coldata <- coldata[match(colnames(count_data), coldata$sample), ]
cat("Sample order check - all match:", all(coldata$sample == colnames(count_data)), "\n")

# Set condition as factor with Control as reference
coldata$condition <- factor(coldata$condition, levels = c("Control", "ASXL1"))

cat("Condition table:\n")
print(table(coldata$condition))

# Create DESeq2 dataset
cat("\nCreating DESeq2 dataset...\n")
dds <- DESeqDataSetFromMatrix(
  countData = count_data,
  colData = coldata,
  design = ~ condition
)

# Pre-filter: remove rows with low counts
keep <- rowSums(counts(dds)) >= 10
dds <- dds[keep, ]
cat("Genes after filtering (rowSums >= 10):", nrow(dds), "\n")

# Run DESeq2
cat("\nRunning DESeq2...\n")
dds <- DESeq(dds)

# Get results
res <- results(dds, contrast = c("condition", "ASXL1", "Control"))
res_df <- as.data.frame(res) %>%
  tibble::rownames_to_column("gene_id")

cat("\nDESeq2 results summary:\n")
summary(res)

# Filter significant DEGs
sig_degs <- res_df %>%
  filter(!is.na(padj) & padj < 0.05)

cat("\nNumber of significant DEGs (padj < 0.05):", nrow(sig_degs), "\n")
cat("Upregulated (log2FC > 0):", sum(sig_degs$log2FoldChange > 0), "\n")
cat("Downregulated (log2FC < 0):", sum(sig_degs$log2FoldChange < 0), "\n")

# Save results
write.csv(res_df, "deseq2_results.csv", row.names=FALSE)
write.csv(sig_degs, "deseq2_sig_degs.csv", row.names=FALSE)
cat("\nResults saved!\n")

EOF

# Load required packages
suppressPackageStartupMessages({
  library(clusterProfiler)
  library(org.Hs.eg.db)
  library(dplyr)
  library(tibble)
})

# Load significant DEGs
cat("Loading significant DEGs...\n")
sig_degs <- read.csv("deseq2_sig_degs.csv")
cat("Number of significant DEGs:", nrow(sig_degs), "\n")
cat("Head of gene IDs:\n")
print(head(sig_degs$gene_id))

# The gene IDs appear to be ENSEMBL IDs with version numbers (e.g., ENSG00000223972.5)
# Remove version numbers for ID mapping
sig_degs$gene_id_clean <- sub("\\..*", "", sig_degs$gene_id)
cat("\nCleaned gene IDs (first 5):\n")
print(head(sig_degs$gene_id_clean))

# Convert ENSEMBL IDs to ENTREZ IDs for enrichGO
cat("\nConverting ENSEMBL IDs to ENTREZ IDs...\n")
entrez_ids <- bitr(sig_degs$gene_id_clean, 
                   fromType = "ENSEMBL", 
                   toType = "ENTREZID", 
                   OrgDb = org.Hs.eg.db)

cat("Number of genes with ENTREZ IDs:", nrow(entrez_ids), "\n")
cat("Conversion rate:", round(nrow(entrez_ids)/length(unique(sig_degs$gene_id_clean))*100, 1), "%\n")

# Run enrichGO analysis - Biological Process
cat("\nRunning enrichGO analysis (BP)...\n")
ego <- enrichGO(gene          = entrez_ids$ENTREZID,
                OrgDb         = org.Hs.eg.db,
                ont           = "BP",
                pAdjustMethod = "BH",
                pvalueCutoff  = 0.05,
                qvalueCutoff  = 0.2,
                readable      = TRUE)

cat("Number of enriched GO terms before simplify:", nrow(ego@result), "\n")

# Apply simplify with similarity > 0.7
cat("\nApplying simplify() with cutoff = 0.7...\n")
ego_simplified <- simplify(ego, cutoff = 0.7, by = "p.adjust", select_fun = min)

cat("Number of enriched GO terms after simplify:", nrow(ego_simplified@result), "\n")

# Save the full simplified results
results_df <- as.data.frame(ego_simplified)
write.csv(results_df, "enrichgo_simplified_results.csv", row.names=FALSE)
cat("\nSimplified results saved!\n")

# Show results related to T cell
cat("\nGO terms related to 'T cell':\n")
tcell_results <- results_df %>%
  filter(grepl("T cell", Description, ignore.case = TRUE)) %>%
  dplyr::select(Description, pvalue, p.adjust, qvalue, Count)
print(tcell_results)

# Create DESeq2 dataset
cat("\nCreating DESeq2 dataset...\n")
dds <- DESeqDataSetFromMatrix(
  countData = count_data,
  colData = coldata,
  design = ~ condition
)

# Pre-filter: remove rows with low counts
keep <- rowSums(counts(dds)) >= 10
dds <- dds[keep, ]
cat("Genes after filtering (rowSums >= 10):", nrow(dds), "\n")

# Run DESeq2

dds <- DESeq(dds)

# Get results

%%bash
cat << 'EOF' > deseq2_analysis.R

# Load required packages
suppressPackageStartupMessages({
  library(DESeq2)
  library(dplyr)
  library(tibble)
})

# Load data
cat("Loading count data and metadata...\n")
count_data <- read.csv("count_data_filtered.csv", row.names=1, check.names=FALSE)
coldata <- read.csv("coldata.csv")

cat("Count data dimensions:", dim(count_data), "\n")
cat("Metadata dimensions:", dim(coldata), "\n")

# Ensure sample order matches
coldata <- coldata[match(colnames(count_data), coldata$sample), ]
cat("Sample order check - all match:", all(coldata$sample == colnames(count_data)), "\n")

# Set condition as factor with Control as reference
coldata$condition <- factor(coldata$condition, levels = c("Control", "ASXL1"))

cat("Condition table:\n")
print(table(coldata$condition))

# Create DESeq2 dataset
cat("\nCreating DESeq2 dataset...\n")
dds <- DESeqDataSetFromMatrix(
  countData = count_data,
  colData = coldata,
  design = ~ condition
)

# Pre-filter: remove rows with low counts
keep <- rowSums(counts(dds)) >= 10
dds <- dds[keep, ]
cat("Genes after filtering (rowSums >= 10):", nrow(dds), "\n")

# Run DESeq2
cat("\nRunning DESeq2...\n")
dds <- DESeq(dds)

# Get results
res <- results(dds, contrast = c("condition", "ASXL1", "Control"))
res_df <- as.data.frame(res) %>%
  tibble::rownames_to_column("gene_id")

cat("\nDESeq2 results summary:\n")
summary(res)

# Filter significant DEGs
sig_degs <- res_df %>%
  filter(!is.na(padj) & padj < 0.05)

cat("\nNumber of significant DEGs (padj < 0.05):", nrow(sig_degs), "\n")
cat("Upregulated (log2FC > 0):", sum(sig_degs$log2FoldChange > 0), "\n")
cat("Downregulated (log2FC < 0):", sum(sig_degs$log2FoldChange < 0), "\n")

# Save results
write.csv(res_df, "deseq2_results.csv", row.names=FALSE)
write.csv(sig_degs, "deseq2_sig_degs.csv", row.names=FALSE)
cat("\nResults saved!\n")

EOF

Rscript deseq2_analysis.R 2>&1

%%bash
cat << 'EOF' > enrichgo_analysis.R

# Load required packages
suppressPackageStartupMessages({
  library(clusterProfiler)
  library(org.Hs.eg.db)
  library(dplyr)
  library(tibble)
})

# Load significant DEGs
cat("Loading significant DEGs...\n")
sig_degs <- read.csv("deseq2_sig_degs.csv")
cat("Number of significant DEGs:", nrow(sig_degs), "\n")
cat("Head of gene IDs:\n")
print(head(sig_degs$gene_id))

# The gene IDs appear to be ENSEMBL IDs with version numbers (e.g., ENSG00000223972.5)
# Remove version numbers for ID mapping
sig_degs$gene_id_clean <- sub("\\..*", "", sig_degs$gene_id)
cat("\nCleaned gene IDs (first 5):\n")
print(head(sig_degs$gene_id_clean))

# Convert ENSEMBL IDs to ENTREZ IDs for enrichGO
cat("\nConverting ENSEMBL IDs to ENTREZ IDs...\n")
entrez_ids <- bitr(sig_degs$gene_id_clean, 
                   fromType = "ENSEMBL", 
                   toType = "ENTREZID", 
                   OrgDb = org.Hs.eg.db)

cat("Number of genes with ENTREZ IDs:", nrow(entrez_ids), "\n")
cat("Conversion rate:", round(nrow(entrez_ids)/length(unique(sig_degs$gene_id_clean))*100, 1), "%\n")

# Run enrichGO analysis - Biological Process
cat("\nRunning enrichGO analysis (BP)...\n")
ego <- enrichGO(gene          = entrez_ids$ENTREZID,
                OrgDb         = org.Hs.eg.db,
                ont           = "BP",
                pAdjustMethod = "BH",
                pvalueCutoff  = 0.05,
                qvalueCutoff  = 0.2,
                readable      = TRUE)

cat("Number of enriched GO terms before simplify:", nrow(ego@result), "\n")

# Apply simplify with similarity > 0.7
cat("\nApplying simplify() with cutoff = 0.7...\n")
ego_simplified <- simplify(ego, cutoff = 0.7, by = "p.adjust", select_fun = min)

cat("Number of enriched GO terms after simplify:", nrow(ego_simplified@result), "\n")

# Save the full simplified results
results_df <- as.data.frame(ego_simplified)
write.csv(results_df, "enrichgo_simplified_results.csv", row.names=FALSE)
cat("\nSimplified results saved!\n")

# Show results related to T cell
cat("\nGO terms related to 'T cell':\n")
tcell_results <- results_df %>%
  filter(grepl("T cell", Description, ignore.case = TRUE)) %>%
  dplyr::select(Description, pvalue, p.adjust, qvalue, Count)
print(tcell_results)

EOF
