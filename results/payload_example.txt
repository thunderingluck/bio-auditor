  colData = sample_info,
  design = ~ Tissue
)

  tibble::rownames_to_column("GeneID") %>%
  dplyr::filter(!is.na(padj))

# Load required packages
suppressPackageStartupMessages({
  library(DESeq2)
  library(dplyr)
  library(tibble)
})

# Load count data and sample info
count_data <- read.csv("control_counts.csv", row.names = 1, check.names = FALSE)
sample_info <- read.csv("sample_info.csv")

cat("Count matrix dimensions:", nrow(count_data), "genes x", ncol(count_data), "samples\n")
cat("Sample info dimensions:", nrow(sample_info), "x", ncol(sample_info), "\n")
cat("Sample info:\n")
print(sample_info)

# Ensure sample order matches between count matrix and sample info
rownames(sample_info) <- sample_info$sample
sample_info <- sample_info[colnames(count_data), ]
cat("\nSample info after reordering:\n")
print(sample_info)

# Set Tissue as factor with baseline_blood as reference
sample_info$Tissue <- factor(sample_info$Tissue, levels = c("baseline_blood", "final_blood"))
cat("\nTissue factor levels:", levels(sample_info$Tissue), "\n")

# Create DESeq2 dataset
dds <- DESeqDataSetFromMatrix(
  countData = count_data,
  colData = sample_info,
  design = ~ Tissue
)

cat("\nDESeq2 dataset created.\n")
cat("Number of genes:", nrow(dds), "\n")

# Run DESeq2
dds <- DESeq(dds)
cat("\nDESeq2 analysis complete.\n")

# Extract results: final_blood vs baseline_blood
res <- results(dds, contrast = c("Tissue", "final_blood", "baseline_blood"))
cat("\nResults summary:\n")
summary(res)

# Convert to data frame
res_df <- as.data.frame(res) %>%
  tibble::rownames_to_column("GeneID") %>%
  dplyr::filter(!is.na(padj))

cat("\nNumber of genes with valid padj:", nrow(res_df), "\n")

# Apply filters: FDR < 0.05, |log2FC| > 1, baseMean >= 10
sig_genes <- res_df %>%
  dplyr::filter(padj < 0.05, abs(log2FoldChange) > 1, baseMean >= 10)

cat("\n=== FINAL RESULTS ===\n")
cat("Number of genes with FDR < 0.05, |log2FC| > 1, and baseMean >= 10:", nrow(sig_genes), "\n")

# Further breakdown
cat("\nBreakdown:\n")
cat("  Upregulated (log2FC > 1):", sum(sig_genes$log2FoldChange > 1), "\n")
cat("  Downregulated (log2FC < -1):", sum(sig_genes$log2FoldChange < -1), "\n")

# Save significant genes
write.csv(sig_genes, "significant_genes.csv", row.names = FALSE)
write.csv(res_df, "all_deseq2_results.csv", row.names = FALSE)

cat("\nTop 20 significant genes:\n")
top20 <- sig_genes %>% dplyr::arrange(padj) %>% head(20)
print(top20[, c("GeneID", "baseMean", "log2FoldChange", "padj")])

%%bash
cat << 'EOF' > run_deseq2.R
# Load required packages
suppressPackageStartupMessages({
  library(DESeq2)
  library(dplyr)
  library(tibble)
})

# Load count data and sample info
count_data <- read.csv("control_counts.csv", row.names = 1, check.names = FALSE)
sample_info <- read.csv("sample_info.csv")

cat("Count matrix dimensions:", nrow(count_data), "genes x", ncol(count_data), "samples\n")
cat("Sample info dimensions:", nrow(sample_info), "x", ncol(sample_info), "\n")
cat("Sample info:\n")
print(sample_info)

# Ensure sample order matches between count matrix and sample info
rownames(sample_info) <- sample_info$sample
sample_info <- sample_info[colnames(count_data), ]
cat("\nSample info after reordering:\n")
print(sample_info)

# Set Tissue as factor with baseline_blood as reference
sample_info$Tissue <- factor(sample_info$Tissue, levels = c("baseline_blood", "final_blood"))
cat("\nTissue factor levels:", levels(sample_info$Tissue), "\n")

# Create DESeq2 dataset
dds <- DESeqDataSetFromMatrix(
  countData = count_data,
  colData = sample_info,
  design = ~ Tissue
)

cat("\nDESeq2 dataset created.\n")
cat("Number of genes:", nrow(dds), "\n")

# Run DESeq2
dds <- DESeq(dds)
cat("\nDESeq2 analysis complete.\n")

# Extract results: final_blood vs baseline_blood
res <- results(dds, contrast = c("Tissue", "final_blood", "baseline_blood"))
cat("\nResults summary:\n")
summary(res)

# Convert to data frame
res_df <- as.data.frame(res) %>%
  tibble::rownames_to_column("GeneID") %>%
  dplyr::filter(!is.na(padj))

cat("\nNumber of genes with valid padj:", nrow(res_df), "\n")

# Apply filters: FDR < 0.05, |log2FC| > 1, baseMean >= 10
sig_genes <- res_df %>%
  dplyr::filter(padj < 0.05, abs(log2FoldChange) > 1, baseMean >= 10)

cat("\n=== FINAL RESULTS ===\n")
cat("Number of genes with FDR < 0.05, |log2FC| > 1, and baseMean >= 10:", nrow(sig_genes), "\n")

# Further breakdown
cat("\nBreakdown:\n")
cat("  Upregulated (log2FC > 1):", sum(sig_genes$log2FoldChange > 1), "\n")
cat("  Downregulated (log2FC < -1):", sum(sig_genes$log2FoldChange < -1), "\n")

# Save significant genes
write.csv(sig_genes, "significant_genes.csv", row.names = FALSE)
write.csv(res_df, "all_deseq2_results.csv", row.names = FALSE)

cat("\nTop 20 significant genes:\n")
top20 <- sig_genes %>% dplyr::arrange(padj) %>% head(20)
print(top20[, c("GeneID", "baseMean", "log2FoldChange", "padj")])
