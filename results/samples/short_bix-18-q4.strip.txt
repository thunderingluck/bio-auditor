### Cell 3 [code]

```
# Calculate descriptive statistics for circularity in rhlR- mutant
rhlr_circularity = rhlr_data['Circularity']

n = len(rhlr_circularity)
mean_val = rhlr_circularity.mean()
std_val = rhlr_circularity.std()  # uses ddof=1 by default (sample std)
sem_val = std_val / np.sqrt(n)

print("=== Circularity Statistics for rhlR- Mutant ===")
print(f"N:                {n}")
print(f"Mean:             {mean_val:.6f}")
print(f"Std Dev (sample): {std_val:.6f}")
print(f"SEM:              {sem_val:.6f}")
print(f"\nUsing scipy for verification:")
from scipy import stats
sem_scipy = stats.sem(rhlr_circularity)
print(f"SEM (scipy):      {sem_scipy:.6f}")
```
